# Atenção ao editar: grupos de robots.txt NÃO são cumulativos. O robô obedece
# apenas ao grupo mais específico que casa com o seu nome, e ignora o grupo "*"
# por completo. Por isso as regras abaixo são repetidas em cada grupo — antes,
# "User-agent: Googlebot / Allow: /" fazia o Googlebot ignorar o Disallow do
# painel admin.
#
# /admin/login fica liberado de propósito: a página tem <meta name="robots"
# content="noindex">, e o Google só consegue ler essa tag se puder rastreá-la.

Allow: /
Allow: /admin/login
Disallow: /admin/
